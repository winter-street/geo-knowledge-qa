```python
"""NER 数据清洗 + 8:1:1 划分"""
import os
import random
from collections import Counter

OUTPUT_DIR = os.path.join(os.path.dirname(__file__), "..", "output")
VALID_LABELS = {"O", "B-SPATIAL", "I-SPATIAL", "B-SURVEY", "I-SURVEY",
                "B-POLICY", "I-POLICY", "B-FUNCTION", "I-FUNCTION",
                "B-INDICATOR", "I-INDICATOR"}

def read_conll(path):
    """读取 CONLL 文件，返回句子列表（每句是 [(char, label), ...]）"""
    sentences = []
    current = []
    with open(path, encoding="utf-8") as f:
        for line in f:
            line = line.strip()
            if not line:
                if current:
                    sentences.append(current)
                    current = []
            else:
                parts = line.split()
                if len(parts) == 2:
                    current.append((parts[0], parts[1]))
    if current:
        sentences.append(current)
    return sentences

def clean_labels(sentences):
    """剔除噪声标签：B-O → O, B-DEGREE → O, I-DEGREE → O"""
    cleaned = []
    removed = 0
    for sent in sentences:
        new_sent = []
        for char, label in sent:
            if label not in VALID_LABELS:
                new_sent.append((char, "O"))
                removed += 1
            else:
                new_sent.append((char, label))
        cleaned.append(new_sent)
    return cleaned, removed

def write_conll(sentences, path):
    with open(path, "w", encoding="utf-8") as f:
        for sent in sentences:
            for char, label in sent:
                f.write(f"{char} {label}\n")
            f.write("\n")

def count_entities(sentences):
    """统计 B- 标签数量"""
    counter = Counter()
    for sent in sentences:
        for _, label in sent:
            if label.startswith("B-"):
                counter[label[2:]] += 1
    return counter

def main():
    src = os.path.join(OUTPUT_DIR, "ner_final.conll")
    sentences = read_conll(src)
    print(f"原始: {len(sentences)} 句")

    # 清洗
    sentences, removed = clean_labels(sentences)
    print(f"清洗噪声标签: {removed} 个 → O")

    # 统计
    entities = count_entities(sentences)
    print(f"清洗后实体分布: {dict(entities)}")
    total = sum(entities.values())
    print(f"总实体数: {total}")

    # 8:1:1 划分（打乱后按比例切）
    random.seed(42)
    random.shuffle(sentences)
    n = len(sentences)
    n_train = int(n * 0.8)
    n_val = int(n * 0.1)
    train = sentences[:n_train]
    val = sentences[n_train:n_train + n_val]
    test = sentences[n_train + n_val:]

    print(f"\n划分: train={len(train)} val={len(val)} test={len(test)}")

    # 写出
    for name, data in [("train", train), ("val", val), ("test", test)]:
        path = os.path.join(OUTPUT_DIR, f"{name}.conll")
        write_conll(data, path)
        ent = count_entities(data)
        print(f"  {name}: {dict(ent)} (总 {sum(ent.values())})")

    # 验证：划分后无标签泄漏
    all_labels = set()
    for s in train + val + test:
        for _, l in s:
            all_labels.add(l)
    unexpected = all_labels - VALID_LABELS
    if unexpected:
        print(f"WARNING: 发现意外标签 {unexpected}")
    else:
        print("\n标签集验证通过 ✓")

if __name__ == "__main__":
    main()
```

```python
"""BERT-NER 微调"""
import os
import sys
import torch
from torch.utils.data import Dataset, DataLoader
from transformers import BertForTokenClassification, BertTokenizerFast, AdamW
from seqeval.metrics import classification_report

os.environ.setdefault("HF_ENDPOINT", "https://hf-mirror.com")

OUTPUT_DIR = os.path.join(os.path.dirname(__file__), "..", "output")
MODEL_DIR = os.path.join(os.path.dirname(__file__), "..", "models", "bert-ner")
LABEL_LIST = ["O", "B-SPATIAL", "I-SPATIAL", "B-SURVEY", "I-SURVEY",
              "B-POLICY", "I-POLICY", "B-FUNCTION", "I-FUNCTION",
              "B-INDICATOR", "I-INDICATOR"]
LABEL2ID = {l: i for i, l in enumerate(LABEL_LIST)}
ID2LABEL = {i: l for l, i in LABEL2ID.items()}
IGNORE_LABEL_ID = -100

class NERDataset(Dataset):
    def __init__(self, conll_path, tokenizer, max_len=128):
        self.sentences = self._read_conll(conll_path)
        self.tokenizer = tokenizer
        self.max_len = max_len

    def _read_conll(self, path):
        sentences, current = [], []
        with open(path, encoding="utf-8") as f:
            for line in f:
                line = line.strip()
                if not line:
                    if current:
                        sentences.append(current)
                        current = []
                else:
                    parts = line.split()
                    if len(parts) == 2:
                        current.append((parts[0], parts[1]))
        if current:
            sentences.append(current)
        return sentences

    def __len__(self):
        return len(self.sentences)

    def __getitem__(self, idx):
        chars, labels = zip(*self.sentences[idx])
        chars, labels = list(chars), list(labels)

        # 截断
        chars = chars[:self.max_len]
        labels = labels[:self.max_len]

        # 中文 BERT：每个字 = 一个 token，天然对齐
        encoding = self.tokenizer(
            chars,
            is_split_into_words=True,
            padding="max_length",
            truncation=True,
            max_length=self.max_len,
            return_tensors="pt"
        )

        # 标签对齐：padding 位置标 -100
        label_ids = []
        for i, label in enumerate(labels):
            if i < self.max_len:
                label_ids.append(LABEL2ID.get(label, LABEL2ID["O"]))
        # padding
        while len(label_ids) < self.max_len:
            label_ids.append(IGNORE_LABEL_ID)

        encoding["labels"] = torch.tensor([label_ids])
        return {k: v.squeeze(0) for k, v in encoding.items()}

def main():
    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    print(f"Device: {device}")

    # 加载 tokenizer
    tokenizer = BertTokenizerFast.from_pretrained("bert-base-chinese")

    # 加载数据
    train_ds = NERDataset(os.path.join(OUTPUT_DIR, "train.conll"), tokenizer)
    val_ds = NERDataset(os.path.join(OUTPUT_DIR, "val.conll"), tokenizer)
    print(f"Train: {len(train_ds)} 句, Val: {len(val_ds)} 句")

    train_loader = DataLoader(train_ds, batch_size=16, shuffle=True)
    val_loader = DataLoader(val_ds, batch_size=16)

    # 模型
    model = BertForTokenClassification.from_pretrained(
        "bert-base-chinese",
        num_labels=len(LABEL_LIST),
        id2label=ID2LABEL,
        label2id=LABEL2ID
    ).to(device)

    # 优化器
    optimizer = AdamW(model.parameters(), lr=5e-5)

    # 训练
    EPOCHS = 3
    for epoch in range(EPOCHS):
        model.train()
        total_loss = 0
        for batch in train_loader:
            batch = {k: v.to(device) for k, v in batch.items()}
            outputs = model(**batch)
            loss = outputs.loss
            loss.backward()
            optimizer.step()
            optimizer.zero_grad()
            total_loss += loss.item()

        avg_loss = total_loss / len(train_loader)
        print(f"Epoch {epoch+1}/{EPOCHS} — loss: {avg_loss:.4f}")

    # 保存
    os.makedirs(MODEL_DIR, exist_ok=True)
    model.save_pretrained(MODEL_DIR)
    tokenizer.save_pretrained(MODEL_DIR)
    print(f"\n模型已保存到 {MODEL_DIR}")

if __name__ == "__main__":
    main()
```

```python
"""BERT-NER 评估：seqeval 分层达标"""
import os
import torch
from transformers import BertForTokenClassification, BertTokenizerFast
from seqeval.metrics import classification_report, f1_score

os.environ.setdefault("HF_ENDPOINT", "https://hf-mirror.com")

MODEL_DIR = os.path.join(os.path.dirname(__file__), "..", "models", "bert-ner")
TEST_PATH = os.path.join(os.path.dirname(__file__), "..", "output", "test.conll")
LABEL_LIST = ["O", "B-SPATIAL", "I-SPATIAL", "B-SURVEY", "I-SURVEY",
              "B-POLICY", "I-POLICY", "B-FUNCTION", "I-FUNCTION",
              "B-INDICATOR", "I-INDICATOR"]
ID2LABEL = {i: l for i, l in enumerate(LABEL_LIST)}
MAIN_TYPES = {"SPATIAL", "SURVEY"}

def read_conll(path):
    sentences, current = [], []
    with open(path, encoding="utf-8") as f:
        for line in f:
            line = line.strip()
            if not line:
                if current:
                    sentences.append(current)
                    current = []
            else:
                parts = line.split()
                if len(parts) == 2:
                    current.append((parts[0], parts[1]))
    if current:
        sentences.append(current)
    return sentences

def main():
    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    print(f"Device: {device}")

    # 加载模型
    tokenizer = BertTokenizerFast.from_pretrained(MODEL_DIR)
    model = BertForTokenClassification.from_pretrained(MODEL_DIR).to(device)
    model.eval()

    # 加载测试集
    sentences = read_conll(TEST_PATH)
    print(f"测试集: {len(sentences)} 句")

    # 推理
    true_labels, pred_labels = [], []
    with torch.no_grad():
        for chars, labels in sentences:
            encoding = tokenizer(
                list(chars),
                is_split_into_words=True,
                padding=True,
                truncation=True,
                return_tensors="pt"
            ).to(device)

            outputs = model(**encoding)
            preds = outputs.logits.argmax(-1)[0].cpu().tolist()

            # 对齐：只取实际字符位置（跳过 padding）
            true_seq, pred_seq = [], []
            for i, (char, label) in enumerate(zip(chars, labels)):
                if i < len(preds):
                    true_seq.append(label)
                    pred_seq.append(ID2LABEL.get(preds[i], "O"))

            true_labels.append(true_seq)
            pred_labels.append(pred_seq)

    # 报告
    print("\n=== 分类报告 ===")
    report = classification_report(true_labels, pred_labels)
    print(report)

    # 分层判定
    # 提取主力类 F1（从 seqeval 报告解析，或直接算）
    from seqeval.metrics import f1_score as seq_f1
    # 主力类过滤：只保留 SPATIAL 和 SURVEY 相关标签
    main_true, main_pred = [], []
    for true_seq, pred_seq in zip(true_labels, pred_labels):
        mt, mp = [], []
        for t, p in zip(true_seq, pred_seq):
            t_type = t[2:] if t.startswith(("B-", "I-")) else None
            if t_type in MAIN_TYPES or t_type is None:
                mt.append(t)
                mp.append(p)
            else:
                mt.append("O")
                mp.append("O")
        main_true.append(mt)
        main_pred.append(mp)

    main_f1 = seq_f1(main_true, main_pred)
    print(f"\n主力类 (SPATIAL+SURVEY) F1: {main_f1:.4f}")

    if main_f1 >= 0.70:
        print("✓ 达标 (≥ 0.70)")
    elif main_f1 < 0.5:
        print("✗ 降级线 (< 0.5) — 建议退回 LLM 方案")
    else:
        print(f"△ 中间地带 (0.5-0.70) — 可用但需改进")

if __name__ == "__main__":
    main()
```

```python
@app.route("/ner", methods=["POST"])
def ner_extract():
    if ner_model is None:
        return jsonify({"error": "NER model not loaded", "entities": []}), 503

    data = request.get_json(force=True)
    text = data.get("text", "")
    if not text:
        return jsonify({"error": "missing 'text' field", "entities": []}), 400

    import torch
    ID2LABEL = ner_model.config.id2label

    chars = list(text)
    encoding = ner_tokenizer(
        chars,
        is_split_into_words=True,
        padding=True,
        truncation=True,
        return_tensors="pt"
    )

    with torch.no_grad():
        outputs = ner_model(**encoding)
        preds = outputs.logits.argmax(-1)[0].tolist()

    # 解析实体
    entities = []
    current_entity = None
    for i, (char, pred_id) in enumerate(zip(chars, preds)):
        if i >= len(preds):
            break
        label = ID2LABEL.get(pred_id, "O")

        if label.startswith("B-"):
            # 保存上一个实体
            if current_entity:
                entities.append(current_entity)
            entity_type = label[2:]
            current_entity = {
                "entity": char,
                "type": entity_type,
                "start": i,
                "end": i + 1
            }
        elif label.startswith("I-") and current_entity:
            entity_type = label[2:]
            if entity_type == current_entity["type"]:
                current_entity["entity"] += char
                current_entity["end"] = i + 1
            else:
                # 类型不匹配，结束当前实体
                entities.append(current_entity)
                current_entity = None
        else:
            if current_entity:
                entities.append(current_entity)
                current_entity = None

    if current_entity:
        entities.append(current_entity)

    return jsonify({"entities": entities})
```

```python
"""bge-small-zh-v1.5 编码 121 chunk 写回 SQLite"""
import os
import sys
import sqlite3
import numpy as np

os.environ.setdefault("HF_ENDPOINT", "https://hf-mirror.com")

DB_PATH = os.path.join(os.path.dirname(__file__), "..", "output", "geo_knowledge.db")

def main():
    from sentence_transformers import SentenceTransformer

    # 连接数据库
    conn = sqlite3.connect(DB_PATH)
    cursor = conn.cursor()

    # 读取所有 chunk
    cursor.execute("SELECT id, text FROM chunks ORDER BY id")
    rows = cursor.fetchall()
    print(f"读取 {len(rows)} 个 chunk")

    if not rows:
        print("WARNING: chunks 表为空")
        return

    ids, texts = zip(*rows)

    # 加载模型
    print("加载 bge-small-zh-v1.5...")
    model = SentenceTransformer("BAAI/bge-small-zh-v1.5")
    dim = model.get_sentence_embedding_dimension()
    print(f"向量维度: {dim}")

    # 编码（bge 检索模式：chunk 侧不加前缀）
    print("编码中...")
    embeddings = model.encode(list(texts), normalize_embeddings=True, show_progress_bar=True)
    print(f"编码完成: {embeddings.shape}")

    # 写回 SQLite
    for i, (chunk_id, vec) in enumerate(zip(ids, embeddings)):
        vec_blob = vec.astype(np.float32).tobytes()
        cursor.execute("UPDATE chunks SET vector = ? WHERE id = ?", (vec_blob, chunk_id))

    # 标记向量类型
    cursor.execute("""
        CREATE TABLE IF NOT EXISTS meta (
            key TEXT PRIMARY KEY,
            value TEXT
        )
    """)
    cursor.execute("INSERT OR REPLACE INTO meta (key, value) VALUES (?, ?)",
                   ("vector_type", "bge-small-zh-v1.5"))
    cursor.execute("INSERT OR REPLACE INTO meta (key, value) VALUES (?, ?)",
                   ("vector_dim", str(dim)))

    conn.commit()
    conn.close()

    print(f"\n✓ {len(ids)} 个向量已写入 SQLite (dim={dim})")

if __name__ == "__main__":
    main()
```

```python
# --- BGE 向量检索 ---
BGE_QUERY_PREFIX = "为这个句子生成表示以用于检索相关文章："
bge_model = None
chunk_vectors = None  # numpy 矩阵 (121, dim)
chunk_meta = []       # [(doc_title, page, text), ...]

def load_bge_index():
    global bge_model, chunk_vectors, chunk_meta
    from sentence_transformers import SentenceTransformer
    import numpy as np

    bge_model = SentenceTransformer("BAAI/bge-small-zh-v1.5")
    dim = bge_model.get_sentence_embedding_dimension()

    conn = sqlite3.connect(SQLITE_DB_PATH)
    cursor = conn.cursor()
    cursor.execute("""
        SELECT c.text, d.title, c.page
        FROM chunks c JOIN documents d ON c.doc_id = d.id
        ORDER BY c.id
    """)
    rows = cursor.fetchall()

    # 从 SQLite 读取预存向量
    cursor.execute("SELECT vector FROM chunks ORDER BY id")
    vec_rows = cursor.fetchall()
    conn.close()

    vectors = []
    for (vec_blob,) in vec_rows:
        if vec_blob:
            vec = np.frombuffer(vec_blob, dtype=np.float32)
            vectors.append(vec)

    if vectors:
        chunk_vectors = np.vstack(vectors)
        # 归一化（确保点积=余弦）
        norms = np.linalg.norm(chunk_vectors, axis=1, keepdims=True)
        chunk_vectors = chunk_vectors / norms
    else:
        chunk_vectors = np.array([])

    chunk_meta = [(text, title, page) for text, title, page in rows]
    print(f"BGE 索引已加载: {len(chunk_meta)} chunks, dim={dim}")

# 根据配置决定是否加载
if RETRIEVAL_MODE == "bge":
    load_bge_index()
```

```python
@app.route("/search", methods=["POST"])
def search():
    data = request.get_json(force=True)
    query = data.get("query", "")
    top_k = data.get("top_k", 5)

    if RETRIEVAL_MODE == "bge" and bge_model is not None and chunk_vectors is not None:
        return search_bge(query, top_k)
    else:
        return search_tfidf(query, top_k)  # 保留原有 TF-IDF 逻辑

def search_bge(query, top_k):
    import numpy as np

    # bge 检索模式：查询侧加前缀
    query_with_prefix = BGE_QUERY_PREFIX + query
    q_vec = bge_model.encode([query_with_prefix], normalize_embeddings=True)[0]

    # 批量点积（已归一化，点积=余弦）
    scores = chunk_vectors @ q_vec

    # Top-K
    top_indices = np.argsort(scores)[::-1][:top_k]

    results = []
    for idx in top_indices:
        text, title, page = chunk_meta[idx]
        results.append({
            "doc_title": title,
            "page": page,
            "text": text,
            "score": float(scores[idx])
        })

    # 闲聊阈值：bge 相似度普遍 0.3-0.7，阈值需重标
    # 先用 0.4 作为临时值，后续实测调整
    CHATTER_THRESHOLD = 0.4
    is_chatter = len(results) == 0 or results[0]["score"] < CHATTER_THRESHOLD

    return jsonify({
        "results": results,
        "is_chatter": is_chatter,
        "mode": "bge"
    })
```
